}

export function vendorName(vendorId: string) {
  return vendorById.get(vendorId)?.name ?? 'Unknown vendor'
}

export function materialName(materialId: string) {
  return materialById.get(materialId)?.item ?? 'Unknown material'
}

export function severityForPercent(value: number): Severity {
  if (value < 75) return 'critical'
  if (value < 90) return 'warning'
  return 'good'
}
